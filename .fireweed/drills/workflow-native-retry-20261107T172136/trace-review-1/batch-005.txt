+		c.Document.Frontmatter["run"] = newRun()

+	}

+	if e = renderChange(info.Root, c); e != nil {

+		return model.State{}, e

+	}

+	projected, e := project.Project(info.Root, c.Slug)

+	if e != nil {

+		_ = renderOriginal(info.Root, c.Path, original)

+		if strings.Contains(e.Error(), "stale base") {

+			return model.State{}, coded("P05", "%v", e)

+		}

+		return model.State{}, e

+	}

+	c, e = parseChange(info.Root, c)

+	if e != nil {

+		return model.State{}, e

+	}

+	cl := claim{Owner: info.Root, Slug: c.Slug, Path: c.Path, Run: frontString(c.Document, "run"), Phase: frontString(c.Document, "phase"), Base: info.Head, Snapshot: projected, Approved: c.Document}

+	if cl.Run == "" {

+		cl.Run = newRun()

+		c.Document.Frontmatter["run"] = cl.Run

+		cl.Approved = c.Document

+		_ = renderChange(info.Root, c)

+	}

+	oid, e := saveClaim(info.Root, cl, "")

+	if e != nil {

+		_ = renderOriginal(info.Root, c.Path, original)

+		return model.State{}, coded("P02", "claim already held: update-ref %v", e)

+	}

+	s := state(info.Root, c, cl, oid)

+	if baseErr == nil && base != "" {

+		if _, e := gitstore.ReadAt(info.Root, base, c.Path); e != nil {

+			if _, e2 := gitstore.ReadAt(info.Root, "main", c.Path); e2 != nil {

+				s.Diagnostics = append(s.Diagnostics, project.Diagnostic{Code: "P04", Message: "change absent from local and shared base", Warning: true})

+			}

+		}

+	}

+	if wasUnstarted && frontBool(c.Document, "regenerate") {

+		if e = deleteDisposable(info.Root, projected, c.Module); e != nil {

+			return s, e

+		}

+		if !newerPassingDrill(info.Root, c.Module) {

+			s.Diagnostics = append(s.Diagnostics, project.Diagnostic{Code: "P10", Message: "no passing drill after latest implementation", Warning: true})

+		}

+	}

+	return s, nil

+}

+func renderOriginal(root, path string, d contracts.Document) error {

+	s, e := contracts.RenderDocument(d)

+	if e != nil {

+		return e

+	}

+	return writeText(root, path, s)

+}

+func unapprovedCount(root string, s project.Snapshot) (int, string, error) {

+	baseName := frontConfig(s, "base", "main")

+	remote := frontConfig(s, "remote", "origin")

+	base, e := gitstore.BaseRef(root, remote, baseName)

+	if e != nil {

+		return 0, "", e

+	}

+	paths, e := filepath.Glob(filepath.Join(root, "changes/archive/*.md"))

+	if e != nil {

+		return 0, base, e

+	}

+	n := 0

+	for _, p := range paths {

+		rel, _ := filepath.Rel(root, p)

+		if _, e := gitstore.ReadAt(root, base, filepath.ToSlash(rel)); e != nil {

+			n++

+		}

+	}

+	return n, base, nil

+}

+func frontConfig(s project.Snapshot, key, def string) string {

+	if v := s.Config[key]; v != nil {

+		return fmt.Sprint(v)

+	}

+	return def

+}

+func deleteDisposable(root string, s project.Snapshot, name string) error {

+	for _, m := range s.Modules {

+		if m.Name != name {

+			continue

+		}

+		return filepath.WalkDir(filepath.Join(root, m.Dir), func(path string, d os.DirEntry, e error) error {

+			if e != nil {

+				return e

+			}

+			if d.IsDir() {

+				return nil

+			}

+			rel, _ := filepath.Rel(filepath.Join(root, m.Dir), path)

+			if contracts.ResolveZone(filepath.ToSlash(rel), m.Zones) == "disposable" {

+				return os.Remove(path)

+			}

+			return nil

+		})

+	}

+	return nil

+}

+func newerPassingDrill(root, module string) bool {

+	refs, _ := gitRefList(root, "refs/fireweed/drills/"+module)

+	return len(refs) > 0

+}
