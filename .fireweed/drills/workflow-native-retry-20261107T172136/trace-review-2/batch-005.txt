+func restoreAttemptFile(root, base, attempt, current, path string) (restoredFile, bool, error) {
+	before, beforeErr := gitstore.ReadAt(root, base, path)
+	saved, savedErr := gitstore.ReadAt(root, attempt, path)
+	now, nowErr := gitstore.ReadAt(root, current, path)
+	if beforeErr == nil && savedErr == nil && bytes.Equal(before, saved) {
+		return restoredFile{}, false, nil
+	}
+	if beforeErr != nil && savedErr != nil {
+		return restoredFile{}, false, nil
+	}
+	if (beforeErr != nil && nowErr != nil) || (beforeErr == nil && nowErr == nil && bytes.Equal(before, now)) {
+		return restoredFile{data: saved, remove: savedErr != nil}, true, nil
+	}
+	if savedErr == nil && nowErr == nil && bytes.Equal(saved, now) {
+		return restoredFile{}, false, nil
+	}
+	if beforeErr != nil || savedErr != nil || nowErr != nil {
+		return restoredFile{}, false, fmt.Errorf("P01: attempt restoration conflicts with later change to %s", path)
+	}
+	dir, e := os.MkdirTemp("", "fireweed-attempt-merge-*")
+	if e != nil {
+		return restoredFile{}, false, e
+	}
+	defer os.RemoveAll(dir)
+	for name, data := range map[string][]byte{"current": now, "base": before, "saved": saved} {
+		if e = os.WriteFile(filepath.Join(dir, name), data, 0600); e != nil {
+			return restoredFile{}, false, e
+		}
+	}
+	cmd := exec.Command("git", "merge-file", "-p", filepath.Join(dir, "current"), filepath.Join(dir, "base"), filepath.Join(dir, "saved"))
+	merged, e := cmd.Output()
+	if e != nil {
+		return restoredFile{}, false, fmt.Errorf("P01: attempt restoration conflicts with later change to %s: %w", path, e)
+	}
+	return restoredFile{data: merged}, true, nil
